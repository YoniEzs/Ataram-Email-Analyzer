# Pinned to the versions the test suite is actually run against, so what is
# verified locally is what ships. Each bump below the divider closes a known CVE
# in the version this file previously pinned.

# Flask and Extensions
Flask==3.1.2
# Pinned explicitly rather than left to Flask's dependency resolution. Was
# unpinned, which allowed <3.0.6: CVE-2024-49767 (multipart resource
# exhaustion — this app takes file uploads) and CVE-2024-34069 (debugger RCE).
Werkzeug==3.1.3
# Was 4.0.0: CVE-2024-6221 (Access-Control-Allow-Private-Network exposing
# internal services) and CVE-2024-1681 (log injection).

python-dotenv==1.2.2

# Email Parsing
extract-msg==0.56.0

# DNS and Network
dnspython==2.8.0
# Was 2.31.0: CVE-2024-35195 (a verify=False call makes the whole session skip
# verification) and CVE-2024-47081 (.netrc credentials leaked to an
# attacker-chosen host).
requests==2.32.5

# WHOIS
python-whois==0.9.6

# HTML Parsing
beautifulsoup4==4.13.5
lxml==6.0.2

# Security
# Was 41.0.7: CVE-2023-49083 (NULL dereference), CVE-2024-0727 (pkcs12 DoS),

cryptography==50.0.0

# Production Server
# Was 21.2.0: CVE-2024-1135 — gunicorn mishandles Transfer-Encoding, permitting
# request smuggling past the front-end proxy. This app sits behind nginx behind
# Cloudflare, which is precisely the topology that CVE targets.

# Rate Limiting
Flask-Limiter==4.1.1

# deployment enforces roughly 4x the configured rate and forgets everything on
# restart. deploy/docker-compose.yml runs Redis and sets
# RATELIMIT_STORAGE_URI=redis://redis:6379/0.
redis==5.2.1
